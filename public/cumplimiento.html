async function cargarJerarquia() { const response = await fetch('/api/jerarquia-cumplimiento'); // 🔥 tu endpoint nuevo const data = await response.json(); renderTablaJerarquia(data); } function renderTablaJerarquia(data) { const header = document.getElementById('headerJerarquia'); const body = document.getElementById('bodyJerarquia'); header.innerHTML = ''; body.innerHTML = ''; if (!data || data.length === 0) return; // 🔍 detectar niveles dinámicamente const niveles = Object.keys(data[0]).filter(k => k.startsWith('nivel_')); const headers = [...niveles, 'indicador', 'cumplimiento']; // 🧱 construir header headers.forEach(h => { const th = document.createElement('th'); th.textContent = h.replace('nivel_', 'Nivel '); header.appendChild(th); }); // 🧠 función de color function getColorClass(valor) { if (valor >= 90) return 'text-success fw-bold'; if (valor >= 70) return 'text-warning fw-bold'; if (valor >= 50) return 'text-primary fw-bold'; return 'text-danger fw-bold'; } // 🧱 construir filas data.forEach(row => { const tr = document.createElement('tr'); niveles.forEach(nivel => { const td = document.createElement('td'); const nombre = row[nivel]?.nombre || ''; const cumplimiento = row[nivel]?.cumplimiento; if (nombre) { td.innerHTML = ` ${nombre}
${cumplimiento?.toFixed(1) || '-'}% `; } tr.appendChild(td); }); // indicador const tdIndicador = document.createElement('td'); tdIndicador.textContent = row.indicador || ''; tr.appendChild(tdIndicador); // cumplimiento indicador const tdCumplimiento = document.createElement('td'); tdCumplimiento.innerHTML = ` ${row.cumplimiento?.toFixed(1) || '-'}% `; tr.appendChild(tdCumplimiento); body.appendChild(tr); }); } // 🚀 inicializar document.addEventListener('DOMContentLoaded', cargarJerarquia);
